let
 Source = Csv.Document(File.Contents("C:\REPLACE_PATH\location_reviews_raw.csv"),[Delimiter=",",Encoding=65001,QuoteStyle=QuoteStyle.Csv]),
 Headers = Table.PromoteHeaders(Source,[PromoteAllScalars=true]),
 CleanText = Table.TransformColumns(Headers,{
  {"Region", each if _=null then "Unknown" else Text.Proper(Text.Trim(_)), type text},
  {"Review_Status", each if _=null then "Unknown" else Text.Proper(Text.Trim(_)), type text}
 }),
 AddCoordinateFlag = Table.AddColumn(CleanText,"Invalid_Coordinates",each [Latitude]=null or [Longitude]=null or [Longitude]=0,type logical)
in AddCoordinateFlag
