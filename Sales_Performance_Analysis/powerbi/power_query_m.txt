let
 Source = Csv.Document(File.Contents("C:\REPLACE_PATH\sales_data_raw.csv"),[Delimiter=",",Encoding=65001,QuoteStyle=QuoteStyle.Csv]),
 Headers = Table.PromoteHeaders(Source,[PromoteAllScalars=true]),
 CleanText = Table.TransformColumns(Headers,{
  {"Region", each if _=null then "Unknown" else Text.Proper(Text.Trim(_)), type text},
  {"Category", each if _=null then "Unknown" else Text.Proper(Text.Trim(_)), type text}
 }),
 AddInvalidQuantity = Table.AddColumn(CleanText,"Invalid_Quantity",each [Quantity]<=0,type logical)
in AddInvalidQuantity
